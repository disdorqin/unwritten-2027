# UNWRITTEN 2027 首发网页终端交接

> 给 Thomas 转发。待全部批次通过 Muse 联审并收到完整 40 位 SHA 后操作。

请在腾讯云 Lighthouse 控制台打开这台服务器的网页终端，使用有 Docker 权限的账户。将下面第一行的 `PIN` 替换为 Thomas 提供的完整 SHA，再把整段一起粘贴到 Ubuntu 终端执行。脚本会在提交切换前核对 GitHub 最新集成分支，随后从公网核对域名和部署版本。如当前终端对 GitHub 私有仓库未配置授权，请回传报错，由仓库管理员走已授权的仓库访问流程。账号密码、令牌和私钥都留在自己的设备上。

切换后看到 `/healthz 原文` 和 `验收钉版` 两处输出时，请逐字符核对 SHA，然后按提示输入同一个完整 SHA。请将 `/healthz` 原文、首页和小镇页两条 `200`、证书状态/有效期、两条 SHA 记录与旧镜像状态回传 Thomas。如果报错，请把失败行和回滚结果一起发回。

```bash
PIN='替换为通过联审的完整40位SHA' ACTION=deploy bash <<'TOWN_PREVIEW'
set -Eeuo pipefail
repo=/opt/apps/unwritten-2027
site=town.disdorqin.cn
branch=town-blockout-integration
origin=https://github.com/disdorqin/unwritten-2027.git
image=unwritten-town-preview
pin=$PIN
action=$ACTION
export GIT_TERMINAL_PROMPT=0 TOWN_RELEASE="$pin" TOWN_PREVIEW_SITE="$site"

die() { printf '%s\n' "$*" >&2; exit 1; }
[[ $pin =~ ^[0-9a-f]{40}$ ]] || die '请填入联审通过的完整 40 位 SHA。'
[[ $action = deploy || $action = rollback ]] || die 'ACTION 只接受 deploy 或 rollback。'
for cmd in git docker curl openssl flock timeout; do command -v "$cmd" >/dev/null; done
mkdir -p /opt/apps
exec 9>/opt/apps/.unwritten-town-preview.lock
flock -n 9 || die '已有预览版发布操作正在运行。'
if [[ ! -e $repo ]]; then
  [[ $action = deploy ]] || die '尚无部署工作区，不能回滚。'
  git clone --no-checkout --single-branch --branch "$branch" "$origin" "$repo"
fi
remote=$(git -C "$repo" remote get-url origin)
case $remote in
  https://github.com/disdorqin/unwritten-2027|https://github.com/disdorqin/unwritten-2027.git|git@github.com:disdorqin/unwritten-2027.git|ssh://git@github.com/disdorqin/unwritten-2027.git) ;;
  *) die '部署目录的 origin 与约定仓库不符。' ;;
esac

record=$repo/.deployed-sha
prior=$repo/.deployed-sha.prev
recorded=''
if [[ -f $record ]]; then
  recorded=$(<"$record")
  [[ $recorded =~ ^[0-9a-f]{40}$ ]] || die '.deployed-sha 格式有误。'
fi
if [[ $action = deploy ]]; then
  git -C "$repo" fetch --no-tags origin "refs/heads/$branch"
  head=$(git -C "$repo" rev-parse --verify 'FETCH_HEAD^{commit}')
  [[ $pin = "$head" ]] || die '验收 SHA 与最新集成分支不一致。请重新联审并取得新 SHA。'
else
  [[ -f $prior && $(<"$prior") = "$pin" ]] || die '回滚 SHA 和 .deployed-sha.prev 不一致。'
  [[ $recorded =~ ^[0-9a-f]{40}$ && $recorded != "$pin" ]] || die '现役 SHA 缺失，或已在目标版本。'
  docker image inspect "$image:$pin" >/dev/null || die '指定回滚镜像不在本机。'
fi
tree="/opt/apps/unwritten-2027-release-$pin"
if [[ ! -e $tree ]]; then git -C "$repo" worktree add --detach "$tree" "$pin"; fi
[[ $(git -C "$tree" rev-parse HEAD) = "$pin" && -z $(git -C "$tree" status --porcelain) ]] \
  || die '独立检出版本不符或包含本地改动。'
cd "$tree"
dc() { docker compose -p unwritten-preview -f tools/compose.town-preview.yaml "$@"; }
dc config --quiet
active=$(dc ps -q web)
[[ $active != *$'\n'* ]] || die '发现多个现役 web 容器。'
previous=$recorded
if [[ -n $active ]]; then
  live=$(docker inspect --format '{{index .Config.Labels "org.opencontainers.image.revision"}}' "$active")
  [[ $live =~ ^[0-9a-f]{40}$ ]] || die '现役 web 容器缺完整 SHA 标记。'
  [[ -z $recorded || $recorded = "$live" ]] || die '现役镜像与部署记录不一致。'
  previous=$live
  live_image=$(docker inspect --format '{{.Image}}' "$active")
  tag_image=$(docker image inspect --format '{{.Id}}' "$image:$live")
  [[ $live_image = "$tag_image" ]] || die '现役容器镜像与其 SHA 标签不一致。'
fi
if [[ -n $previous ]]; then
  docker image inspect "$image:$previous" >/dev/null
  printf '保留旧镜像 %s:%s\n' "$image" "$previous"
fi

state=$(mktemp -d "$repo/.unwritten-deploy.XXXXXX")
if [[ -f $record ]]; then cp "$record" "$state/record.backup"; fi
if [[ -f $prior ]]; then cp "$prior" "$state/prior.backup"; fi
switched=0
ledger_started=0
done_ok=0

check_http() {
  local expected=$1 path code
  curl --fail --silent --show-error --connect-timeout 5 --max-time 12 \
    --retry 10 --retry-all-errors --retry-delay 2 \
    --output "$state/healthz" "https://$site/healthz" || return 1
  if ! cmp -s "$state/healthz" <(printf 'town-preview %s' "$expected") &&
     ! cmp -s "$state/healthz" <(printf 'town-preview %s\n' "$expected"); then
    printf 'healthz 版本不符或包含额外内容。预期 %s\n' "$expected" >&2
    return 1
  fi
  printf '\n/healthz 原文\n'
  cat "$state/healthz"
  printf '\n'
  for path in / /town-blockout.html; do
    code=$(curl --fail --silent --show-error --connect-timeout 5 --max-time 12 \
      --output /dev/null --write-out '%{http_code}' "https://$site$path") || return 1
    printf '%s %s\n' "$code" "$path"
    [[ $code = 200 ]] || return 1
  done
}
cert() {
  timeout 10 openssl s_client -connect "$site:443" -servername "$site" </dev/null 2>/dev/null |
    openssl x509 -noout -issuer -dates -fingerprint -sha256
}
restore_record() {
  local target=$1 backup=$2
  if [[ -f $backup ]]; then mv -f "$backup" "$target"; else rm -f "$target"; fi
}
on_exit() {
  local code=$?
  set +e
  trap - EXIT
  if (( ! done_ok && switched )); then
    if [[ -n $previous && $previous != "$pin" ]]; then
      printf '发布未确认，尝试恢复 %s\n' "$previous" >&2
      if TOWN_RELEASE="$previous" dc up -d --no-build --pull never && check_http "$previous"; then
        printf '已回滚至 %s\n' "$previous" >&2
      else
        printf '自动回滚未核验通过，请保留现有镜像和证书卷并回报。\n' >&2
      fi
    else
      dc stop web
      printf '没有可切回的现役版本，已请求停止 web。\n' >&2
    fi
  fi
  if (( ! done_ok && ledger_started )); then
    restore_record "$record" "$state/record.backup"
    restore_record "$prior" "$state/prior.backup"
  fi
  rm -rf -- "$state"
  exit "$code"
}
trap on_exit EXIT
trap 'exit 130' INT
trap 'exit 143' TERM
cert >"$state/cert-before" || :
if [[ $action = deploy ]]; then
  if [[ -n $active && $previous = "$pin" ]]; then
    printf '目标 SHA 已在运行，核验现有服务。\n'
  else
    dc build web
    docker image inspect "$image:$pin" >/dev/null
  fi
  latest=$(git -C "$repo" ls-remote --exit-code origin "refs/heads/$branch")
  [[ $latest = "$pin"$'\t'"refs/heads/$branch" ]] \
    || die '构建期间远端分支发生变化，未切换。请重新联审。'
fi
if [[ -z $active || $previous != "$pin" ]]; then
  switched=1
  dc up -d --no-build --pull never
fi
dc ps
check_http "$pin"
cert >"$state/cert-after"
if [[ -s $state/cert-before ]] && cmp -s "$state/cert-before" "$state/cert-after"; then
  printf 'TLS 证书未变，沿用原证书。\n'
elif [[ -s $state/cert-before ]]; then
  printf 'TLS 证书已变化。\n'
else
  printf '已取得 TLS 证书，切换前无可比记录。\n'
fi
cat "$state/cert-after"
printf '验收钉版 %s\n请将上面的 /healthz SHA 与钉版逐字符核对，确认后输入完整 SHA。\n' "$pin"
IFS= read -r confirmed </dev/tty
[[ $confirmed = "$pin" ]] || die '人工确认与验收钉版不符。'

ledger_started=1
if [[ -n $previous && $previous != "$pin" ]]; then
  printf '%s\n' "$previous" >"$state/previous"
  mv -f "$state/previous" "$prior"
fi
printf '%s\n' "$pin" >"$state/current"
mv -f "$state/current" "$record"
done_ok=1
printf '.deployed-sha %s\n' "$pin"
if [[ -f $prior ]]; then
  prev_pin=$(<"$prior")
  printf '.deployed-sha.prev %s\n' "$prev_pin"
  if docker image inspect "$image:$prev_pin" >/dev/null; then
    printf '上一版本镜像可用。\n'
  else
    printf '上一版本镜像缺失。\n'
  fi
else
  printf '无上一版部署记录。\n'
fi
TOWN_PREVIEW
```

需回滚时，将第一行 `ACTION=deploy` 换为 `ACTION=rollback`，`PIN` 填当前 `.deployed-sha.prev` 指定的完整 SHA。此分支直接使用现有 SHA 镜像，并重复 HTTPS 和人工确认。保留两个记录和镜像，执行任何切换时保留 Compose 的证书卷。首次上线若无旧版，切换失败后预览会停止，保留诊断状态和证书。

`town-preview` 模式在当前 Vite 配置中只打包小镇页面，独立酒馆入口及其生成步骤与本次构建无关。证书签发失败时先核对 DNS A 记录与 80/443 连通性。
