```bash
git diff --no-ext-diff --exit-code 551100a -- \
  src/shared/mirror-lake.ts \
  tools/voxel-tavern/town-blockout/mirror-lake-zone-sample.ts &&
node --import tsx --input-type=module <<'NODE'
import assert from 'node:assert/strict';
import { isMirrorLakeIce, mirrorLake } from './src/shared/mirror-lake.ts';
import { signedIceClearance } from './tools/voxel-tavern/town-blockout/mirror-lake-zone-sample.ts';

const original = [[74, 185], [71, 210], [71, 160], [58, 231], [58, 139], [26, 185]];
const drawing = [[73.65, 185], [70.70, 209.65], [70.70, 160.35], [58, 231], [58, 139], [26.35, 185]];
const [lakeE, lakeN] = mirrorLake.center;

function probe(id, [east, north]) {
  let offIce = 0;
  let nearest = Infinity;
  let nearestAngle;
  for (let degree = 0; degree < 360; degree++) {
    const angle = degree * Math.PI / 180;
    const point = [east + 3 * Math.cos(angle), north + 3 * Math.sin(angle)];
    if (!isMirrorLakeIce(...point)) offIce++;
    const clearance = signedIceClearance(point);
    if (clearance < nearest) {
      nearest = clearance;
      nearestAngle = degree;
    }
  }
  return { id, east, north, offIce, nearest, nearestAngle };
}

function inward45cm([east, north]) {
  const distance = Math.hypot(lakeE - east, lakeN - north);
  return [east + 0.45 * (lakeE - east) / distance, north + 0.45 * (lakeN - north) / distance];
}

const before = original.map((center, i) => probe(`F${i + 1}`, center));
const after = drawing.map((center, i) => probe(`F${i + 1}`, center));
const alternatives = [1, 2].flatMap(i => {
  const exact = inward45cm(original[i]);
  return [
    probe(`F${i + 1} 径向 0.45 m`, exact),
    probe(`F${i + 1} 径向圆整`, exact.map(axis => Number(axis.toFixed(2)))),
  ];
});
assert.deepEqual(before.map(row => row.offIce), [47, 54, 54, 0, 0, 47]);
assert.ok([...after, ...alternatives].every(row => row.offIce === 0));
console.table(before);
console.table(after);
console.table(alternatives);
NODE
```
