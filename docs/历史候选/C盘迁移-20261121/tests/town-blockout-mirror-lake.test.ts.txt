import assert from 'node:assert/strict';
import test from 'node:test';
import { isMirrorLakeIce, mirrorLake } from '../src/shared/mirror-lake';
import { buildMirrorLakeScene, mirrorFishingPropAnchors } from '../tools/voxel-tavern/town-blockout/mirror-lake-scene';
import { mirrorDrillAllowed, mirrorEastEntries, mirrorFishingSites } from '../tools/voxel-tavern/town-blockout/mirror-lake-site';
import { canEnter, createTerrainGrid } from '../tools/voxel-tavern/town-blockout/terrain';

const grid = createTerrainGrid();

test('Mirror Lake keeps all approved workstations and each fishing cue on the ice sheet', () => {
  assert.deepEqual(mirrorLake, { center: [50, 185], radius: [27, 55], height: 10 });
  const mirrorScene = buildMirrorLakeScene(grid.heightAt);
  const { group } = mirrorScene;
  assert.equal(group.children.filter(child => child.name === 'mirror-lake-fishing-props').length, 1);
  for (const site of mirrorFishingSites) {
    assert.equal(grid.heightAt(site.x, site.n), mirrorLake.height, site.id);
    assert.equal(grid.surfaces[site.n * grid.gridWidth + site.x], 4, site.id);
    assert.equal(mirrorDrillAllowed(site.x, site.n), false, site.id);
    const { hole, float, bubble, fishOrbit } = mirrorFishingPropAnchors(site);
    for (const { east, north } of [hole, float, bubble, fishOrbit]) {
      assert.ok(isMirrorLakeIce(east, north), site.id);
      assert.equal(grid.heightAt(east, north), mirrorLake.height, site.id);
    }
  }
  const [centerEast, centerNorth] = mirrorLake.center;
  assert.equal(mirrorDrillAllowed(centerEast, centerNorth), true);
  assert.equal(mirrorDrillAllowed(50, 241), false);
  assert.equal(grid.surfaces[66 * grid.gridWidth + 344], 4);
  assert.equal(grid.heightAt(344, 66), 5.5);
  mirrorScene.dispose();
  assert.equal(group.children.length, 0);
  assert.equal(group.userData.colliders.length, 0);
});

test('each east entry joins existing carriageway grade and the Mirror Lake ice datum', () => {
  for (const { id, north, roadX, roadY, iceX } of mirrorEastEntries) {
    assert.ok(Math.abs(grid.heightAt(roadX, north) - roadY) < 1e-6, id);
    assert.equal(grid.heightAt(iceX, north), mirrorLake.height, id);
    const steps = Math.ceil((roadX - iceX) / .08);
    let previousHeight = grid.heightAt(roadX, north);
    for (let i = 1; i <= steps; i++) {
      const east = roadX + (iceX - roadX) * i / steps;
      assert.ok(canEnter(east, north), id);
      const height = grid.heightAt(east, north);
      assert.ok(Math.abs(height - previousHeight) <= .43, id);
      previousHeight = height;
    }
  }
});
