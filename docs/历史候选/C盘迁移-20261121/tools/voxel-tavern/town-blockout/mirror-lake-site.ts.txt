import { isMirrorLakeIce, mirrorLake, mirrorLakeSignedDistance } from '../../../src/shared/mirror-lake';
import layout from './layout.json';

export const mirrorFishingSites = [
  { id: 'F1', x: 74, n: 185 },
  { id: 'F2', x: 71, n: 210 },
  { id: 'F3', x: 71, n: 160 },
  { id: 'F4', x: 58, n: 231 },
  { id: 'F5', x: 58, n: 139 },
  { id: 'F6', x: 26, n: 185 },
] as const;

export const mirrorFishingOccupancyRadius = 3;
export const mirrorFishingDrillSpacing = 8;

export function mirrorDrillAllowed(east: number, north: number) {
  return isMirrorLakeIce(east, north)
    && mirrorFishingSites.every(site => Math.hypot(east - site.x, north - site.n) >= mirrorFishingDrillSpacing);
}

// East entries use the signed west shoulder of the existing west-loop. The drawing
// doesn't provide entry elevations, so interpolate the corresponding surveyed road.
const westLoop = layout.roads.find(road => road.id === 'west-loop')!;
function westLoopAt(north: number) {
  const index = westLoop.points.findIndex(([, n]) => n <= north);
  const [ax, an, ay] = westLoop.points[index - 1];
  const [bx, bn, by] = westLoop.points[index];
  const fraction = (an - north) / (an - bn);
  return {
    x: ax + (bx - ax) * fraction,
    y: ay + (by - ay) * fraction,
    west: ax + (bx - ax) * fraction - westLoop.width / 2 * Math.hypot(1, (bx - ax) / (bn - an)),
  };
}

export const mirrorEastEntries = mirrorFishingSites.slice(0, 3).map(site => {
  const road = westLoopAt(site.n);
  return { id: site.id, north: site.n, roadX: road.x, shoulderX: road.west, roadY: road.y, iceX: site.x };
});

export const mirrorLakeServices = [
  { id: 'fishing-lodge', name: '冰钓登记处', x: 93, n: 174, width: 6, depth: 8 },
  { id: 'shore-shelter', name: '湖畔亭', x: 90, n: 196, width: 5, depth: 6 },
  { id: 'curling-shed', name: '冰壶器材棚', x: 86, n: 148, width: 4, depth: 6 },
].map(service => ({ ...service, ground: westLoopAt(service.n).y }));

const smooth = (value: number) => {
  const t = Math.min(1, Math.max(0, value));
  return t * t * (3 - 2 * t);
};
const mix = (a: number, b: number, t: number) => a + (b - a) * t;

// Apply this locally after road grading. The east ramps must touch the legacy
// carriageway without replacing its grade. Deep ice remains flat past the lip.
export function shapeMirrorEastShore(east: number, north: number, elevation: number, roadEdge: number) {
  if (east < 63 || east > 103 || north < 139 || north > 217 || roadEdge < 0) return elevation;

  if (mirrorLakeSignedDistance(east, north) <= 0) elevation = mirrorLake.height;
  for (const entry of mirrorEastEntries) {
    const fromAxis = Math.abs(north - entry.north);
    if (fromAxis > 3 || east < entry.iceX || east > entry.roadX) continue;
    const fraction = Math.min(1, (east - entry.iceX) / (entry.shoulderX - entry.iceX));
    const target = mix(mirrorLake.height, entry.roadY, fraction);
    elevation = mix(elevation, target, 1 - smooth((fromAxis - 1) / 2));
  }

  for (const site of mirrorLakeServices) {
    const doorX = site.x - site.width / 2;
    const eastWall = site.x + site.width / 2;
    const roadX = westLoopAt(site.n).x;
    const dx = Math.max(0, roadX - east, east - eastWall);
    const dz = Math.max(0, Math.abs(north - site.n) - site.depth / 2);
    const doorHeight = east >= doorX ? site.ground : westLoopAt(north).y;
    elevation = mix(elevation, doorHeight, 1 - smooth(Math.hypot(dx, dz) / 2));
  }
  return elevation;
}

export function mirrorEastShoreKeepClear(east: number, north: number, margin: number) {
  const entry = mirrorEastEntries.some(ramp => east > ramp.iceX - margin && east < ramp.roadX + margin
    && Math.abs(north - ramp.north) < 3 + margin);
  return entry || mirrorLakeServices.some(site => {
    const roadX = westLoopAt(site.n).x;
    return east > roadX - margin && east < site.x + site.width / 2 + margin
      && Math.abs(north - site.n) < site.depth / 2 + margin;
  });
}
