type RouteId = keyof typeof rallyCourses;
type StorageSlot = Pick<Storage, 'getItem' | 'setItem'>;

export type RallySplit = {
  index: number;
  cumulativeMs: number;
  segmentMs: number;
  targetMs: number;
  differenceMs: number;
  withinTarget: boolean;
};
type RallyResult = { totalMs: number; splits: readonly RallySplit[] };

function readBest(storage: StorageSlot | undefined, key: string) {
  try {
    const ms = Number(storage?.getItem(key));
    if (Number.isSafeInteger(ms) && ms > 0) return ms;
  } catch { /* Scores remain available in the session. */ }
}

export function createMountainRally(route: RouteId, now: () => number, storage?: StorageSlot) {
  const course: RallyCourse = rallyCourses[route];
  const storageKey = `unwritten.mountain-rally.v1.${route}`;
  const timer = createOrderedRaceTimer(course.checkpoints, now);
  let splits: RallySplit[] = [];
  let best = readBest(storage, storageKey);
  let pending: RallyResult | undefined;

  function reset() {
    timer.reset();
    splits = [];
  }

  return {
    update(sledding: boolean, crosses: (checkpoint: RallyCheckpoint) => boolean) {
      if (!sledding) return undefined;
      const passed = timer.pass(crosses);
      if (!passed || passed.index === 0) return undefined;
      const { index, cumulativeMs, segmentMs } = passed;
      const { targetMs } = course.checkpoints[index];
      const differenceMs = segmentMs - targetMs;
      const split: RallySplit = {
        index, cumulativeMs, segmentMs, targetMs, differenceMs,
        withinTarget: Math.abs(differenceMs) <= targetMs * rallySplitTolerance,
      };
      splits.push(split);
      if (passed.finished) {
        pending = { totalMs: cumulativeMs, splits: [...splits] };
        if (best === undefined || cumulativeMs < best) {
          best = cumulativeMs;
          try { storage?.setItem(storageKey, String(best)); } catch { /* The session still records it. */ }
        }
        reset();
      }
      return split;
    },
    reset,
    running: timer.running,
    nextCheckpoint: () => course.checkpoints[timer.nextIndex()],
    best: () => best,
    takeResult() { const result = pending; pending = undefined; return result; },
  };
}
